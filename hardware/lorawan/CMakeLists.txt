cmake_minimum_required(VERSION 3.20)
project(opi_lorawan LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(Git REQUIRED)
set(RADIOLIB_REVISION "3b1d917bbacba0062dd7064c1157893ef9c6384e")
set(RADIOLIB_SOURCE_DIR "" CACHE PATH "Checkout of pinned RadioLib revision")
if(NOT EXISTS "${RADIOLIB_SOURCE_DIR}/src/RadioLib.h")
  message(FATAL_ERROR "Set RADIOLIB_SOURCE_DIR to RadioLib ${RADIOLIB_REVISION}; no automatic download")
endif()
execute_process(COMMAND ${GIT_EXECUTABLE} -C "${RADIOLIB_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND ${GIT_EXECUTABLE} -C "${RADIOLIB_SOURCE_DIR}" status --porcelain --untracked-files=all OUTPUT_VARIABLE dirty OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT revision STREQUAL RADIOLIB_REVISION OR NOT dirty STREQUAL "")
  message(FATAL_ERROR "RadioLib must be the pinned, unmodified revision")
endif()
find_path(WIRINGPI_INCLUDE_DIR wiringPi.h REQUIRED)
find_library(WIRINGPI_LIBRARY NAMES wiringPi REQUIRED)
add_subdirectory("${RADIOLIB_SOURCE_DIR}" radiolib EXCLUDE_FROM_ALL)
add_executable(opi-lorawan main.cpp)
target_include_directories(opi-lorawan PRIVATE "${WIRINGPI_INCLUDE_DIR}")
target_link_libraries(opi-lorawan PRIVATE RadioLib OpenSSL::Crypto "${WIRINGPI_LIBRARY}")
target_compile_options(opi-lorawan PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(lorawan-tests tests.cpp)
  target_link_libraries(lorawan-tests PRIVATE RadioLib OpenSSL::Crypto)
  target_compile_options(lorawan-tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME persistence-and-configuration COMMAND lorawan-tests)
  add_test(NAME cli-offline COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/test_cli.py" $<TARGET_FILE:opi-lorawan>)
  add_executable(lorawan-hal-tests test_hal.cpp)
  target_include_directories(lorawan-hal-tests PRIVATE "${WIRINGPI_INCLUDE_DIR}")
  target_link_libraries(lorawan-hal-tests PRIVATE RadioLib OpenSSL::Crypto)
  target_compile_options(lorawan-hal-tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME hal-with-gpio-doubles COMMAND lorawan-hal-tests)
endif()
install(TARGETS opi-lorawan RUNTIME DESTINATION bin)
install(FILES opi-lorawan.service DESTINATION lib/systemd/system)
install(FILES lorawan.conf.example credentials.conf.example DESTINATION share/opi-lorawan)
install(FILES "${RADIOLIB_SOURCE_DIR}/license.txt" DESTINATION share/licenses/opi-lorawan RENAME RadioLib-MIT.txt)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../LICENSE" DESTINATION share/licenses/opi-lorawan RENAME COPYING)
